Erstelle dynamische Route `src/src/pages/aktuelle-stellen/[slug].astro`:

```astro
---
import { getCollection } from 'astro:content';
import BaseLayout from '../../layouts/BaseLayout.astro';
import HeroTextJobs from '../../components/sections/HeroTextJobs.astro';
import ColumnsWithHeadline from '../../components/sections/ColumnsWithHeadline.astro';
import GridBlocks from '../../components/sections/GridBlocks.astro';
import FullBleedBlocks from '../../components/sections/FullBleedBlocks.astro';

export async function getStaticPaths() {
  const jobs = await getCollection('jobs', ({ data }) => data.published);
  return jobs.map((job) => ({
    params: { slug: job.slug },
    props: { job },
  }));
}

const { job } = Astro.props;
const { Content } = await job.render();

const employmentTypeLabels = {
  vollzeit: 'Vollzeit',
  teilzeit: 'Teilzeit',
  werkstudent: 'Werkstudent',
  praktikum: 'Praktikum',
};
---

<BaseLayout
  title={`${job.data.title} | schlafgut`}
  currentPath={Astro.url.pathname}
>
  <HeroTextJobs
    title={job.data.title}
    gender={job.data.gender}
    type={employmentTypeLabels[job.data.employmentType]}
    location={job.data.location}
  />

  <ColumnsWithHeadline
    headline="Deine Aufgaben"
    background="beige"
  >
    <Content />
  </ColumnsWithHeadline>

  <ColumnsWithHeadline
    headline="Was bringst du mit?"
    background="beige"
  >
    <Fragment set:html={job.data.requirements} />
  </ColumnsWithHeadline>

  <ColumnsWithHeadline
    headline="Warum wir?"
    background="yellow"
  >
    <Fragment set:html={job.data.benefits} />
    <p>Interesse? Dann bewirb Dich und sende deine Unterlagen an:</p>
    <p><a href={`mailto:${job.data.contactEmail}`}>{job.data.contactEmail}</a></p>
    <p><a href={`mailto:${job.data.contactEmail}`} class="cta__button">jetzt bewerben</a></p>
  </ColumnsWithHeadline>

  <!-- Standard Footer Sections from Original -->
  <GridBlocks variant="fullwidth" />
  <FullBleedBlocks />
</BaseLayout>
```

---
import { getCollection } from 'astro:content';
import BaseLayout from '../../layouts/BaseLayout.astro';
import HeroText from '../../components/sections/HeroText.astro';

const jobs = await getCollection('jobs', ({ data }) => data.published);

const employmentTypeLabels = {
  vollzeit: 'Vollzeit',
  teilzeit: 'Teilzeit',
  werkstudent: 'Werkstudent',
  praktikum: 'Praktikum',
};
---

<BaseLayout
  title="Aktuelle Stellen | schlafgut"
  currentPath={Astro.url.pathname}
>
  <HeroText headline="Aktuelle Stellen" />

  <section class="section__job-list" data-background="white">
    <div class="innerwrapper smaller">
      {jobs.length === 0 ? (
        <p class="no-jobs">Aktuell keine offenen Stellen. Schau später wieder vorbei!</p>
      ) : (
        <ul class="job-list">
          {jobs.map((job) => (
            <li class="job-card">
              <a href={`/aktuelle-stellen/${job.slug}/`}>
                <h3>{job.data.title}</h3>
                <p class="job-meta">
                  {job.data.gender} · {employmentTypeLabels[job.data.employmentType]} · {job.data.location}
                </p>
              </a>
            </li>
          ))}
        </ul>
      )}
    </div>
  </section>
</BaseLayout>
